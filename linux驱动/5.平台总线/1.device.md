### 第一步：准备“资产证明” (`struct resource`)

男嘉宾来相亲，不能空口无凭说自己有钱，得拿出实实在在的证明。在 Linux 里，硬件的“资产”主要有两类：**内存地址（寄存器）**和 **中断号（IRQ）**。

内核提供了一个专门的结构体 `struct resource` 来记录这些信息。

假设我们要点亮 RK3568 上的一个 LED，它的 GPIO 寄存器基地址是 `0xFDD60000`，占 32 个字节。我们需要这样定义它的资源：

C

```c
#include <linux/ioport.h>

// 定义资源数组（男嘉宾有几套房，就写几个）
static struct resource led_resources[] = {
    [0] = {
        .start = 0xFDD60000,         // 房产起始地址 (寄存器物理基地址)
        .end   = 0xFDD60000 + 32 - 1, // 房产结束地址
        .flags = IORESOURCE_MEM,     // 资产类型：这是一块内存/寄存器
    },
    // 如果这个硬件还有中断号，可以接着写 [1]
    /*
    [1] = {
        .start = 66,                 // 中断号
        .end   = 66,
        .flags = IORESOURCE_IRQ,     // 资产类型：这是一个中断线
    }
    */
};
```

### 第二步：填写“个人简历” (`struct platform_device`)

资产证明开好了，现在要填男嘉宾的个人简历了。Linux 提供了一个叫 `struct platform_device` 的结构体。

在这里，**最最最重要的是他的名字 (`name`)**，因为以后女嘉宾（Driver）就是靠这个名字来认人的！

同时，内核要求你必须提供一个 `release` 函数（哪怕里面是空的），否则在卸载设备时内核会疯狂弹警告。

C

```c
#include <linux/platform_device.h>

// 1. 写一个释放函数（必须要有，类似于相亲失败或者退出的清理工作）
static void led_device_release(struct device *dev)
{
    printk("LED device 已经从总线上注销啦！\n");
}

// 2. 填写简历（组装 platform_device）
static struct platform_device led_device = {
    .name = "rk3568-myled",           // 【核心】男嘉宾的名字！一字都不能错！
    .id   = -1,                       // 编号（如果系统里只有一个这种设备，填 -1）
    .num_resources = ARRAY_SIZE(led_resources), // 资产证明有几份？
    .resource = led_resources,        // 把第一步的资产证明挂上去
    .dev = {
        .release = led_device_release, // 挂载清理函数
    },
};
```

### 第三步：正式提交给婚介所 (`platform_device_register`)

简历写好了，最后一步就是把它交到系统总线上去。这通常写在这个模块的 `init` 函数里。

C

```c
#include <linux/module.h>

// 模块加载函数：男嘉宾进场
static int __init led_device_init(void)
{
    int ret;
    // 调用内核 API，把这个 device 注册到平台总线上
    ret = platform_device_register(&led_device);
    if (ret) {
        printk("注册男嘉宾失败了！\n");
        return ret;
    }
    
    printk("注册男嘉宾成功，等待匹配...\n");
    return 0;
}

// 模块卸载函数：男嘉宾离场
static void __exit led_device_exit(void)
{
    // 从平台总线上注销
    platform_device_unregister(&led_device);
}

module_init(led_device_init);
module_exit(led_device_exit);
MODULE_LICENSE("GPL");
```
