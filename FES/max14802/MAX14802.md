rt_uint8_t tx_buf[MAX14802_CHIP_COUNT * 2];

    rt_size_t  tx_len = MAX14802_CHIP_COUNT * 2;

    rt_size_t  sent;

    int buf_idx = 0;

    /* 从最后一片到第一片依次填充发送缓冲区 */

 这里写的很好，一个for填充一片的数据，buf_idx一开始是0，然后变成1，然后变成2，然后进第二次for循环 这里 tx_buf >> 8的原因是，spi发的是uint8  & 0xFF是用来把uint16裁切成uint8的
    for (int i = MAX14802_CHIP_COUNT - 1; i >= 0; i--)

    {

        tx_buf[buf_idx++] = (dev->state[i] >> 8) & 0xFF;  /* 高字节 (CH15~CH8) */

        tx_buf[buf_idx++] = dev->state[i] & 0xFF;          /* 低字节 (CH7~CH0)  */

    }

    /* SPI 发送 */

    sent = rt_spi_send(&dev->spi_dev, tx_buf, tx_len);

    if (sent != tx_len)

    {

        rt_kprintf("[MAX14802] SPI send failed: %d/%d\n", sent, tx_len);

        return -RT_ERROR;

    }
