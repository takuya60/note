## 1.滑动平均
直接对这九个点求均值
坏处是会在跳变的时候变成梯形
## 2.去极值平均
去掉最大值与最小值，对剩下的求平均
还是会变成梯形
```c
// 先定义窗口大小
#define channel_num 1
#define BUFF_WIN_SIZE 16
static uint16_t history[channel_num][BUFF_WIN_SIZE];// 这里要设置成static
static uint8_t history_index[channel_num]={0};
static rt_bool histroy_init[channel_num] ={RT_FALSE};
// 消除上电骤变
if (!histroy_init[ch]&& copy_len>0){
	for(int i=0;i<BUFF_WIN_SIZE;i++)
		histroy[ch][i]= wave->data[0];
	history_index[ch]=0;
	history_init [ch]=RT_TRUE;	
}
// 从dma搬运过来的数据里面取值
for (uint32_t k = 0; k < copy_len; k++){
	uint16_t current_val=wave->data[k];
	history[ch][history_index]=current_val;
	history_index=(history_index[ch]+1)%BUFF_WIN_SIZE;
	uint32_t sum=0;
	uint16_t max=0；
	uint16_t min=0xFFFF
	for(int i=0;i<BUFF_WIN_SIZE;i++){
	uint16_t val=history[ch][history_index];
	sum+=val;
	if (val > max_val) max_val = val;
	if (val < min_val) min_val = val;
	}
	sum = sum - max_val - min_val;
	wave->filtered[k] = (uint16_t)(sum / (STIM_ADC_FILTER_WIN - 2));
	
}

```
