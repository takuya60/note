### 1. 设备挂载到了i2c或者spi总线下

总线会自动解析设备树上的中断资源并且填入i2c_client结构体
我们直接拿就行
```c
#include <linux/i2c.h>

struct i2c_client *i2c_device

static int my_sensor_probe(struct i2c_client *client, const struct i2c_device_id *id)
{
    int irq_num;
	i2c_device= client,
    // 内核 I2C 核心层已经帮你从 DTS 解析并映射好了！直接拿来用！
    irq_num = i2c_device
    ->irq; 

    if (irq_num <= 0) {
        dev_err(&client->dev, "Failed to get IRQ from DTS\n");
        return -EINVAL;
    }

    dev_info(&client->dev, "Got mapped Virtual IRQ: %d\n", irq_num);

    // 接下来就可以直接调用 request_threaded_irq(irq_num, ...) 了
    // ...
    return 0;
}
```
### 2.通过 Platform API 获取

```c
#include <linux/platform_device.h>

static int my_device_probe(struct platform_device *pdev)
{
    int irq_num;

    // 参数 0 表示获取该节点下的第 1 个中断资源
    irq_num = platform_get_irq(pdev, 0);

    if (irq_num < 0) {
        dev_err(&pdev->dev, "No IRQ resource found\n");
        return irq_num; // 返回错误码
    }

    // 拿到了虚拟中断号，准备申请中断...
    return 0;
}
```
