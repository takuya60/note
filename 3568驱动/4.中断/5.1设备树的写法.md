// 通常在 rk3568-xxx.dts 文件中
&i2c1 {
    status = "okay";
    
    // 你的传感器节点
    my_sensor: sensor@44 {
        compatible = "myvendor,my-sensor";
        reg = <0x44>; // I2C 设备地址
        
        /* === 中断相关的核心配置 === */
        
        // 1. 指定中断控制器 (中断的父节点)
        // 因为连接的是普通 GPIO，所以父节点是该 GPIO 所在的 bank (gpio0)
        interrupt-parent = <&gpio0>;
        
        // 2. 指定中断号和触发类型
        // 格式: <引脚号 触发标志位>
        // RK_PB5 是瑞芯微定义的引脚宏，通常等于 13 (8*1 + 5)
        // IRQ_TYPE_EDGE_FALLING 表示下降沿触发
        interrupts = <RK_PB5 IRQ_TYPE_EDGE_FALLING>;
        
        // 可选：有些严格的驱动会要求显式声明 GPIO，用于复位或电源控制
        // irq-gpios = <&gpio0 RK_PB5 GPIO_ACTIVE_LOW>;
    };
};
