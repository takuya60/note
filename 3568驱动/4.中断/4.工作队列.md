```c
#include <linux/module.h>
#include <linux/interrupt.h>
#include <linux/workqueue.h>
#include <linux/delay.h>

// 1. 定义驱动私有结构体
struct my_sensor_data {
    int irq_gpio;
    int data_value;
    struct work_struct sensor_work; // 嵌套 work_struct
};

struct my_sensor_data my_dev;

// 2. 工作队列的处理函数（下半部，可以睡眠）
static void sensor_work_func(struct work_struct *work)
{
    // 核心：利用 container_of 从 work 指针获取 my_dev 的指针
    struct my_sensor_data *dev = container_of(work, struct my_sensor_data, sensor_work);
    
    printk("Bottom Half execution starts. Reading I2C...\n");
    // 模拟耗时或可能睡眠的 I2C 操作
    msleep(100); 
    dev->data_value = 1024; // 假设读到的值
    
    printk("Bottom Half finished. Data: %d\n", dev->data_value);
}

// 3. 中断处理函数（上半部，不可睡眠）
static irqreturn_t sensor_irq_handler(int irq, void *dev_id)
{
    struct my_sensor_data *dev = (struct my_sensor_data *)dev_id;
    
    printk("Top Half: IRQ triggered! Scheduling work...\n");
    
    // 调度工作队列（将任务推入内核共享工作线程去异步执行）
    schedule_work(&dev->sensor_work);
    
    // 快速返回，恢复硬件中断响应
    return IRQ_HANDLED;
}

static int __init my_sensor_init(void)
{
    // 初始化工作队列
    INIT_WORK(&my_dev.sensor_work, sensor_work_func);
    
    // 假设申请中断 (伪代码，省略了 GPIO 到 IRQ 的转换)
    // request_irq(irq_num, sensor_irq_handler, IRQF_TRIGGER_FALLING, "my_sensor", &my_dev);
    
    return 0;
}
```
