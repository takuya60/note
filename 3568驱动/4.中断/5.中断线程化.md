#include <linux/module.h>
#include <linux/interrupt.h>
#include <linux/delay.h>
#include <linux/slab.h>

// 1. 定义设备私有结构体
struct sensor_data {
    int irq_num;
    int temperature;
    // 可能还有其他设备相关的指针，比如 i2c_client 等
};

struct sensor_data *my_sensor;

// 2. 顶半部 (Top Half / Hard IRQ) - 硬件中断上下文
// 这里的要求是：极速执行，绝对不能睡眠！
static irqreturn_t sensor_hardirq_handler(int irq, void *dev_id)
{
    struct sensor_data *sensor = (struct sensor_data *)dev_id;
    
    // 在这里通常做两件事：
    // 1. 检查中断是否真的是你的设备产生的（如果是共享中断线）
    // 2. 清除硬件的中断标志位（对于某些需要手动 clear 状态的芯片）
    
    // 假设我们确认是我们的中断源
    printk(KERN_INFO "Hard IRQ triggered! Waking up the thread...\n");
    
    // 核心机制：返回 IRQ_WAKE_THREAD，内核会自动唤醒我们绑定的内核线程去执行下半部
    return IRQ_WAKE_THREAD; 
}

// 3. 底半部 (Bottom Half / Threaded IRQ) - 线程上下文
// 这里的要求是：可以睡眠，可以执行耗时操作，可以使用 Mutex 锁。
static irqreturn_t sensor_threaded_handler(int irq, void *dev_id)
{
    struct sensor_data *sensor = (struct sensor_data *)dev_id;
    
    printk(KERN_INFO "Threaded Handler running. Now we can sleep or do I2C transfers.\n");
    
    // 模拟耗时的 I2C 读取操作 (msleep 会导致线程睡眠，在顶半部是绝对禁止的)
    msleep(50); 
    
    // 假设读取到了数据
    sensor->temperature = 25;
    printk(KERN_INFO "Data read complete: %d C\n", sensor->temperature);
    
    // 处理完毕，返回 IRQ_HANDLED 告诉内核本次中断处理彻底结束
    return IRQ_HANDLED;
}

// 4. 驱动初始化
static int __init my_sensor_init(void)
{
    int ret;
    
    // 分配内存
    my_sensor = kzalloc(sizeof(struct sensor_data), GFP_KERNEL);
    if (!my_sensor) return -ENOMEM;
    
    // 假设我们从设备树(Device Tree)中获取到了中断号
    my_sensor->irq_num = 101; // 示例伪造的IRQ号
    
    // 核心 API：申请线程化中断
    ret = request_threaded_irq(my_sensor->irq_num,
                               sensor_hardirq_handler,   // 顶半部函数 (如果传 NULL，内核会提供一个默认的)
                               sensor_threaded_handler,  // 底半部函数 (在独立的内核线程中运行)
                               IRQF_TRIGGER_FALLING | IRQF_ONESHOT, // 触发标志位
                               "my_i2c_sensor",          // 中断名称 (在 /proc/interrupts 中显示)
                               my_sensor);               // 传递给 handler 的参数 (dev_id)

    if (ret) {
        printk(KERN_ERR "Failed to request threaded IRQ: %d\n", ret);
        kfree(my_sensor);
        return ret;
    }
    
    printk(KERN_INFO "Sensor driver initialized with threaded IRQ.\n");
    return 0;
}

// 5. 驱动卸载
static void __exit my_sensor_exit(void)
{
    // 释放中断
    free_irq(my_sensor->irq_num, my_sensor);
    kfree(my_sensor);
    printk(KERN_INFO "Sensor driver exited.\n");
}

module_init(my_sensor_init);
module_exit(my_sensor_exit);
MODULE_LICENSE("GPL");
